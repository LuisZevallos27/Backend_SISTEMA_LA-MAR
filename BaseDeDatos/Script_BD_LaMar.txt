-- ---------------------------------------------------------------------
-- 1. TABLAS MAESTRAS
-- ---------------------------------------------------------------------

CREATE TABLE Cliente (
    id_cliente        INT IDENTITY(1,1) PRIMARY KEY,
    dni_cliente       VARCHAR(11) NOT NULL,
    nombres_cliente   VARCHAR(60) NOT NULL,
    apellidos_cliente VARCHAR(60) NOT NULL,
    telefono_cliente  VARCHAR(15),
    correo_cliente    VARCHAR(80)
);
GO

CREATE TABLE Mesa (
    id_mesa      INT IDENTITY(1,1) PRIMARY KEY,
    numero_mesa  INT NOT NULL,
    zona_mesa    VARCHAR(20) NOT NULL,
    capacidad    INT NOT NULL,
    estado       VARCHAR(20) NOT NULL,
    CONSTRAINT UQ_Mesa_numero UNIQUE (numero_mesa),
    CONSTRAINT CK_Mesa_zona   CHECK (zona_mesa IN ('Piso 1', 'Piso 2', 'VIP')),
    CONSTRAINT CK_Mesa_estado CHECK (estado IN ('Disponible', 'Reservado', 'Ocupado'))
);
GO

-- password_empleado guarda el HASH BCrypt (nunca la contrasena en texto plano).
-- Queda NULL en los datos de prueba; el backend (DataInitializer) la completa.
CREATE TABLE Empleado (
    id_empleado        INT IDENTITY(1,1) PRIMARY KEY,
    dni_empleado       CHAR(8) NOT NULL,
    password_empleado  VARCHAR(100),
    nombres_empleado   VARCHAR(60) NOT NULL,
    apellidos_empleado VARCHAR(60) NOT NULL,
    cargo              VARCHAR(40) NOT NULL,
    turno              VARCHAR(20) NOT NULL,
    CONSTRAINT UQ_Empleado_dni UNIQUE (dni_empleado),
    CONSTRAINT CK_Empleado_cargo CHECK (cargo IN ('Mozo', 'Recepcionista', 'Administrador'))
);
GO

CREATE TABLE Proveedor (
    id_proveedor       INT IDENTITY(1,1) PRIMARY KEY,
    ruc_proveedor      CHAR(11) NOT NULL,
    prov_razon_social  VARCHAR(100) NOT NULL,
    telefono_proveedor VARCHAR(15),
    CONSTRAINT UQ_Proveedor_ruc UNIQUE (ruc_proveedor)
);
GO

CREATE TABLE Plato (
    id_plato     INT IDENTITY(1,1) PRIMARY KEY,
    nombre_plato VARCHAR(80) NOT NULL,
    categoria    VARCHAR(40),
    precio_plato DECIMAL(10,2) NOT NULL,
    descripcion  VARCHAR(200),
    imagen_url   VARCHAR(255),
    CONSTRAINT CK_Plato_precio CHECK (precio_plato >= 0)
);
GO

CREATE TABLE Insumo (
    id_insumo     INT IDENTITY(1,1) PRIMARY KEY,
    nombre_insumo VARCHAR(80) NOT NULL,
    unidad_medida VARCHAR(15) NOT NULL
);
GO

-- ---------------------------------------------------------------------
-- 2. TABLAS TRANSACCIONALES
-- ---------------------------------------------------------------------

CREATE TABLE Reserva (
    id_reserva    INT IDENTITY(1,1) PRIMARY KEY,
    id_cliente    INT NOT NULL,
    id_mesa       INT NOT NULL,
    canal_reserva VARCHAR(20),
    fecha_reserva DATE,
    hora_reserva  TIME,                 -- obligatoria desde la API
    cant_personas INT,
    estado        VARCHAR(20),
    CONSTRAINT FK_Reserva_Cliente FOREIGN KEY (id_cliente) REFERENCES Cliente(id_cliente),
    CONSTRAINT FK_Reserva_Mesa    FOREIGN KEY (id_mesa)    REFERENCES Mesa(id_mesa),
    CONSTRAINT CK_Reserva_estado  CHECK (estado IN ('Pendiente', 'Confirmada', 'Cancelada', 'Atendida'))
);
GO

CREATE TABLE Pedido (
    id_pedido        INT IDENTITY(1,1) PRIMARY KEY,
    id_cliente       INT NULL,          -- NULL = cliente no registrado
    id_mesa          INT NOT NULL,
    id_empleado      INT NOT NULL,
    fecha_hora       DATETIME,
    tipo_comprobante VARCHAR(15),
    monto_total      DECIMAL(10,2),     -- base imponible + IGV (sin propina)
    estado_pedido    VARCHAR(20),
    CONSTRAINT FK_Pedido_Cliente  FOREIGN KEY (id_cliente)  REFERENCES Cliente(id_cliente),
    CONSTRAINT FK_Pedido_Mesa     FOREIGN KEY (id_mesa)     REFERENCES Mesa(id_mesa),
    CONSTRAINT FK_Pedido_Empleado FOREIGN KEY (id_empleado) REFERENCES Empleado(id_empleado),
    CONSTRAINT CK_Pedido_estado   CHECK (estado_pedido IN ('Abierto', 'Cerrado', 'Anulado')),
    CONSTRAINT CK_Pedido_comprob  CHECK (tipo_comprobante IN ('Boleta', 'Factura'))
);
GO

CREATE TABLE Abastecimiento (
    id_abastecimiento INT IDENTITY(1,1) PRIMARY KEY,
    id_proveedor      INT NOT NULL,
    fecha             DATETIME,
    estado            VARCHAR(20) NOT NULL CONSTRAINT DF_Abast_estado DEFAULT 'Pendiente',
    CONSTRAINT FK_Abast_Proveedor FOREIGN KEY (id_proveedor) REFERENCES Proveedor(id_proveedor),
    CONSTRAINT CK_Abast_estado    CHECK (estado IN ('Pendiente', 'Recibido', 'Anulado'))
);
GO

-- ---------------------------------------------------------------------
-- 3. TABLAS INTERMEDIAS
-- ---------------------------------------------------------------------

CREATE TABLE Detalle_Pedido (
    id_pedido INT NOT NULL,
    id_plato  INT NOT NULL,
    cantidad  INT NOT NULL,
    subtotal  DECIMAL(10,2),
    CONSTRAINT PK_Detalle_Pedido PRIMARY KEY (id_pedido, id_plato),
    CONSTRAINT FK_DetPed_Pedido  FOREIGN KEY (id_pedido) REFERENCES Pedido(id_pedido),
    CONSTRAINT FK_DetPed_Plato   FOREIGN KEY (id_plato)  REFERENCES Plato(id_plato),
    CONSTRAINT CK_DetPed_cant    CHECK (cantidad > 0)
);
GO

CREATE TABLE Receta_Insumo (
    id_plato       INT NOT NULL,
    id_insumo      INT NOT NULL,
    cantidad_usada DECIMAL(8,3) NOT NULL,
    CONSTRAINT PK_Receta_Insumo PRIMARY KEY (id_plato, id_insumo),
    CONSTRAINT FK_Receta_Plato  FOREIGN KEY (id_plato)  REFERENCES Plato(id_plato),
    CONSTRAINT FK_Receta_Insumo FOREIGN KEY (id_insumo) REFERENCES Insumo(id_insumo)
);
GO

CREATE TABLE Detalle_Abastecimiento (
    id_abastecimiento INT NOT NULL,
    id_insumo         INT NOT NULL,
    cantidad          DECIMAL(10,2) NOT NULL,
    costo_total       DECIMAL(10,2) NOT NULL,
    CONSTRAINT PK_Detalle_Abast PRIMARY KEY (id_abastecimiento, id_insumo),
    CONSTRAINT FK_DetAbast_Abast  FOREIGN KEY (id_abastecimiento) REFERENCES Abastecimiento(id_abastecimiento),
    CONSTRAINT FK_DetAbast_Insumo FOREIGN KEY (id_insumo)         REFERENCES Insumo(id_insumo)
);
GO

-- ---------------------------------------------------------------------
-- VERIFICACION (cada consulta debe devolver lo indicado)
-- ---------------------------------------------------------------------
-- 1) 12 mesas, 11 platos, 4 empleados, 3 pedidos

UNION ALL SELECT 'Empleado', COUNT(*) FROM Empleado
UNION ALL SELECT 'Pedido',   COUNT(*) FROM Pedido;
